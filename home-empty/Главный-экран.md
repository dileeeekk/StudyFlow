# Главный экран нового аккаунта

Изменения уже внесены в Android Studio-проект. Повторно вставлять код не нужно.

На экране: настоящее имя из профиля, 0%, нулевые счётчики задач, занятий и заметок, карточка первых шагов и две кнопки. Нижняя навигация сохранена. Данные задач пока не подключены: пустое состояние показывается всем аккаунтам. Это не удаляет данные пользователя.

Карточки будущих задач и занятий сохранены в item_home_task.xml и item_home_class.xml. Демонстрационные надписи в этих шаблонах перенесены в tools:text — они видны только в редакторе. При подключении API потребуется переключать пустое состояние на список реальных данных.

Кнопки пока показывают временную подсказку; создание и сохранение задач не реализовано. Профиль и выход продолжают работать.

## Проверка

В Android Studio выбери app и эмулятор, нажми Run. Войди или зарегистрируйся. На главном экране должны быть нули и настоящее имя пользователя. На телефон обновление можно установить позже, без удаления приложения.

```kotlin
package com.example.studyflowapp

import android.content.Intent
import android.widget.TextView
import android.widget.Toast
import android.graphics.Color
import android.os.Bundle
import android.view.View
import androidx.activity.SystemBarStyle
import androidx.activity.enableEdgeToEdge
import androidx.lifecycle.ViewModelProvider
import com.example.studyflowapp.auth.AuthViewModel
import com.example.studyflowapp.auth.NetworkActivity
import com.google.android.material.dialog.MaterialAlertDialogBuilder
import androidx.core.view.ViewCompat
import androidx.core.view.WindowInsetsCompat
import androidx.core.view.updatePadding

/** Authenticated account; empty dashboard until task and timetable features are connected. */
class HomeActivity : NetworkActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge(
            statusBarStyle = SystemBarStyle.light(Color.TRANSPARENT, Color.TRANSPARENT),
            navigationBarStyle = SystemBarStyle.light(Color.TRANSPARENT, Color.rgb(23, 36, 67))
        )
        setContentView(R.layout.activity_home)
        val root = findViewById<View>(R.id.home_root)
        val auth = ViewModelProvider(this)[AuthViewModel::class.java]
        root.visibility = View.INVISIBLE
        fun returnToLogin() {
            startActivity(Intent(this, LoginActivity::class.java)
                .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK))
            finish()
        }
        auth.state.observe(this) { state ->
            if (state.expired || state.loggedOut) {
                state.error?.let { Toast.makeText(this, it, Toast.LENGTH_LONG).show() }
                returnToLogin()
            } else if (state.user != null) {
                root.visibility = View.VISIBLE
                val user = state.user
                val firstName = user.fullName.trim().split(Regex("\\s+"), limit = 2).first()
                findViewById<TextView>(R.id.home_greeting).text = getString(R.string.fresh_greeting, firstName)
                findViewById<TextView>(R.id.home_avatar).text = firstName.take(1).uppercase()
                findViewById<View>(R.id.home_tab_profile).setOnClickListener {
                    if (auth.state.value?.loading == true) return@setOnClickListener
                    MaterialAlertDialogBuilder(this).setTitle(R.string.auth_account)
                        .setMessage("${user.fullName}\n${user.email}\n${user.university}")
                        .setNegativeButton(R.string.auth_cancel, null)
                        .setPositiveButton(R.string.auth_logout) { _, _ -> auth.logout() }.show()
                }
            } else if (state.error != null) {
                MaterialAlertDialogBuilder(this).setTitle(R.string.auth_offline_title)
                    .setMessage(state.error).setCancelable(false)
                    .setPositiveButton(R.string.auth_retry) { _, _ -> withNetworkAccess { auth.loadProfile(retry = true) } }
                    .setNegativeButton(R.string.auth_logout) { _, _ -> auth.logout() }.show()
            }
        }
        withNetworkAccess { auth.loadProfile() }

        val scroll = findViewById<View>(R.id.home_scroll)
        val navigation = findViewById<View>(R.id.home_navigation)
        
        ViewCompat.setOnApplyWindowInsetsListener(root) { _, insets ->
            val safe = insets.getInsets(WindowInsetsCompat.Type.systemBars() or WindowInsetsCompat.Type.displayCutout())
            scroll.updatePadding(left = safe.left, top = maxOf(safe.top, dp(44)) + dp(16), right = safe.right)
            navigation.updatePadding(left = safe.left, right = safe.right, bottom = maxOf(safe.bottom, dp(34)))

            insets
        }
        ViewCompat.requestApplyInsets(root)
        findViewById<View>(R.id.home_tab_home).isSelected = true
        findViewById<View>(R.id.home_avatar).setOnClickListener {
            findViewById<View>(R.id.home_tab_profile).performClick()
        }
        // UI-only actions: no demonstration tasks are created or saved.
        for ((ids, message) in listOf(
            listOf(R.id.home_add, R.id.home_setup_task, R.id.home_tab_tasks) to R.string.fresh_tasks_later,
            listOf(R.id.home_timetable, R.id.home_setup_classes, R.id.home_tab_schedule) to R.string.fresh_schedule_later,
            listOf(R.id.home_setup_goal) to R.string.fresh_goal_later
        )) {
            ids.forEach { id -> findViewById<View>(id).setOnClickListener {
                Toast.makeText(this, message, Toast.LENGTH_SHORT).show()
            } }
        }
    }

    private fun dp(value: Int): Int = (value * resources.displayMetrics.density + 0.5f).toInt()
}
```

```kotlin
package com.example.studyflowapp

import android.content.Intent
import android.widget.TextView
import android.widget.Toast
import android.graphics.Color
import android.os.Bundle
import android.view.View
import androidx.activity.SystemBarStyle
import androidx.activity.enableEdgeToEdge
import androidx.lifecycle.ViewModelProvider
import com.example.studyflowapp.auth.AuthViewModel
import com.example.studyflowapp.auth.NetworkActivity
import com.google.android.material.dialog.MaterialAlertDialogBuilder
import androidx.core.view.ViewCompat
import androidx.core.view.WindowInsetsCompat
import androidx.core.view.updatePadding

/** Authenticated account; empty dashboard until task and timetable features are connected. */
class HomeActivity : NetworkActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge(
            statusBarStyle = SystemBarStyle.light(Color.TRANSPARENT, Color.TRANSPARENT),
            navigationBarStyle = SystemBarStyle.light(Color.TRANSPARENT, Color.rgb(23, 36, 67))
        )
        setContentView(R.layout.activity_home)
        val root = findViewById<View>(R.id.home_root)
        val auth = ViewModelProvider(this)[AuthViewModel::class.java]
        root.visibility = View.INVISIBLE
        fun returnToLogin() {
            startActivity(Intent(this, LoginActivity::class.java)
                .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK))
            finish()
        }
        auth.state.observe(this) { state ->
            if (state.expired || state.loggedOut) {
                state.error?.let { Toast.makeText(this, it, Toast.LENGTH_LONG).show() }
                returnToLogin()
            } else if (state.user != null) {
                root.visibility = View.VISIBLE
                val user = state.user
                val firstName = user.fullName.trim().split(Regex("\\s+"), limit = 2).first()
                findViewById<TextView>(R.id.home_greeting).text = getString(R.string.auth_greeting, firstName)
                findViewById<View>(R.id.home_tab_profile).setOnClickListener {
                    if (auth.state.value?.loading == true) return@setOnClickListener
                    MaterialAlertDialogBuilder(this).setTitle(R.string.auth_account)
                        .setMessage("${user.fullName}\n${user.email}\n${user.university}")
                        .setNegativeButton(R.string.auth_cancel, null)
                        .setPositiveButton(R.string.auth_logout) { _, _ -> auth.logout() }.show()
                }
            } else if (state.error != null) {
                MaterialAlertDialogBuilder(this).setTitle(R.string.auth_offline_title)
                    .setMessage(state.error).setCancelable(false)
                    .setPositiveButton(R.string.auth_retry) { _, _ -> withNetworkAccess { auth.loadProfile(retry = true) } }
                    .setNegativeButton(R.string.auth_logout) { _, _ -> auth.logout() }.show()
            }
        }
        withNetworkAccess { auth.loadProfile() }

        val scroll = findViewById<View>(R.id.home_scroll)
        val navigation = findViewById<View>(R.id.home_navigation)
        
        ViewCompat.setOnApplyWindowInsetsListener(root) { _, insets ->
            val safe = insets.getInsets(WindowInsetsCompat.Type.systemBars() or WindowInsetsCompat.Type.displayCutout())
            scroll.updatePadding(left = safe.left, top = maxOf(safe.top, dp(44)) + dp(16), right = safe.right)
            navigation.updatePadding(left = safe.left, right = safe.right, bottom = maxOf(safe.bottom, dp(34)))

            insets
        }
        ViewCompat.requestApplyInsets(root)
        findViewById<View>(R.id.home_tab_home).isSelected = true
        // Show the actual date rather than the date in the Figma example.
        findViewById<TextView>(R.id.home_date).text = java.text.SimpleDateFormat(
            "EEEE, MMMM d", java.util.Locale.ENGLISH
        ).format(java.util.Date()).uppercase(java.util.Locale.ENGLISH)
        // UI-only actions: no demonstration tasks are created or saved.
        for ((ids, message) in listOf(
            listOf(R.id.home_add, R.id.home_setup_task, R.id.home_tab_tasks) to R.string.fresh_tasks_later,
            listOf(R.id.home_timetable, R.id.home_setup_classes, R.id.home_tab_schedule) to R.string.fresh_schedule_later,
            listOf(R.id.home_setup_goal) to R.string.fresh_goal_later
        )) {
            ids.forEach { id -> findViewById<View>(id).setOnClickListener {
                Toast.makeText(this, message, Toast.LENGTH_SHORT).show()
            } }
        }
    }

    private fun dp(value: Int): Int = (value * resources.displayMetrics.density + 0.5f).toInt()
}
```
